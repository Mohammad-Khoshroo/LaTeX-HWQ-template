\question#{25}

% \paragraph{دست‌گرمی}
برای هرکدام از زبان‌های زیر یک عبارت منظم \lr{Regular Expression} بنویسید. هر بخش از عبارات منظم خود را توضیح دهید در غیر این صورت به عبارت منظم تنها، نمره‌ای تعلق نمیگیرد.

$$B = \{\text{\lr{0}} , \text{\lr{1}}\}$$
$$L = \{a,b,c,d\}$$
$$K = \{a,b,c\}$$


\begin{itemize}
    \item[\textbf{(a}]
          تمام رشته‌هایی که شامل سه صفر متوالی (\lr{000}) \textbf{نیستند}. ($\Sigma = B$)(5 نمره)
    \item[\textbf{(b}]
          تمام رشته‌هایی که در آن‌ها هیچگاه حرف a بلافاصله قبل از حرف b نیامده است (یعنی فاقد زیررشته \lr{ab} هستند). ($\Sigma = K$) (5 نمره)
    \item[\textbf{(c}]
          تمام رشته‌هایی که حداقل شامل یک حرف a و حداقل شامل یک حرف b باشند. ($\Sigma = L$) (5 نمره)
    \item[\textbf{(d}]
          تمام رشته‌هایی که تعداد \lr{1}های آن‌ها مضربی از 3 باشد. ($\Sigma = B$) (5 نمره)
    \item[\textbf{(e}]
          تمام رشته‌هایی که با یک حرف شروع می‌شوند و دقیقاً با همان حرف نیز خاتمه می‌یابند (طول رشته حداقل ۱ است). ($\Sigma = K$) (5 نمره)

\end{itemize}

\ifcompileanswers\solution

    \begin{itemize}
        \item[\textbf{(a}]
              برای جلوگیری از ایجاد \lr{000}، می‌توانیم رشته را به بلوک‌هایی تقسیم کنیم که حتماً با \lr{1} ختم می‌شوند و قبل از آن حداکثر دو صفر دارند (یعنی \lr{1} یا \lr{01} یا \lr{001}). در انتهای رشته نیز ممکن است حداکثر دو صفر بیاید که با هیچ \lr{1}ای ختم نشود.
              \lr{$$RE = (1 + 01 + 001)^*(\epsilon + 0 + 00)$$}
        \item[\textbf{(b}]
              چون زیررشته \lr{ab} غیرمجاز است، هرگاه حرف a ظاهر شود، بعد از آن فقط می‌تواند حرف a یا c بیاید (یا رشته تمام شود). اگر بخواهیم پس از a دوباره از b استفاده کنیم، حتماً باید یک حرف c واسطه شود (مثل \lr{acb}). پس بلوک‌های ما شامل \lr{b,c} و دنباله‌ای از \lr{a}هاست که به c ختم می‌شوند. در انتها نیز می‌تواند هر تعدادی a بیاید.
              $$RE = (b + c + aa^*c)^*a^*$$
        \item[\textbf{(c}]
              $$RE = \Sigma^* a \Sigma^* b \Sigma^* + \Sigma^* b \Sigma^* a \Sigma^*$$
        \item[\textbf{(d}]
              \lr{$$RE = (0^*10^*10^*10^*)^*$$}
        \item[\textbf{(e}]
              $$RE = a\Sigma^*a + b\Sigma^*b + c\Sigma^*c + a + b + c$$

    \end{itemize}

\fi
